\subsection{Theorem 1: OOD Distance Inversion Breakdown and MSSP Monotonicity Recovery}
\label{subsec:proof_theorem1}

\begin{theorem}[OOD Distance Inversion Breakdown]
\label{thm:distance_inversion}
Let $f_\theta: \mathbb{R}^k \to \mathbb{R}$ be an $L$-layer neural network with piecewise-affine activations (e.g., LeakyReLU with negative slope $\alpha_{\text{leak}} \in (0, 1)$), parameterized by weights $\theta = \{W^{(l)}, b^{(l)}\}_{l=1}^L$. Suppose $f_\theta$ is trained via empirical risk minimization on a bounded compact support $\mathcal{K}_{\text{train}} \subset [0, R_0]^k$ generated by a fixed perturbation radius $\epsilon_0$. Then:
\begin{enumerate}
    \item \textbf{(Inversion Pathology)}: There exists an unbounded polyhedral cone $\mathcal{C}_{\text{out}} \subset \mathbb{R}^k \setminus \mathcal{K}_{\text{train}}$ such that for any ray $d(t) = t \cdot v$ with $v \in \mathbb{R}_+^k$, $\norm{v}_2 = 1$, the directional derivative is strictly negative:
    \begin{equation}
    \label{eq:thm1_inversion}
    \lim_{t \to \infty} \nabla_d f_\theta(d(t)) \cdot v \le -\gamma < 0 \quad (\gamma > 0),
    \end{equation}
    which implies $\lim_{t \to \infty} f_\theta(d(t)) = -\infty$ and $\lim_{t \to \infty} \sigma(f_\theta(d(t))) = 0.0000$.
    \item \textbf{(MSSP Monotonicity Recovery)}: Under Multi-Scale Subspace Perturbation (MSSP) with geometrically progressive scales $\mathcal{S}_{\text{scales}} = \{\epsilon_s\}_{s=1}^S$ spanning the metric domain $[0, R_{\max}]$, if empirical risk satisfies $\mathcal{L}_{\text{train}}(\theta) \le \varepsilon_{\text{opt}}$, then the distance-ranking network maintains strictly positive ranking monotonicity:
    \begin{equation}
    \label{eq:thm1_recovery}
    \frac{\partial f_\theta(d)}{\partial d_j} \ge \kappa > 0 \quad \forall j \in \{1, \dots, k\}, \quad \forall d \in [0, R_{\max}]^k,
    \end{equation}
    guaranteeing that $\sigma(f_\theta(d_{\text{att}})) \ge \sigma(f_\theta(d_{\text{nom}})) + \Delta_{\text{margin}}$ for all out-of-distribution volumetric attacks.
\end{enumerate}
\end{theorem}

\begin{IEEEproof}
We establish each part sequentially:

\textbf{Part 1: Proof of Inversion Pathology in Canonical LUNAR}.
An $L$-layer neural network with LeakyReLU activations partitions the input space $\mathbb{R}^k$ into a finite set of $P$ convex polyhedral cells $\{\mathcal{C}_p\}_{p=1}^P$~\cite{arora2018understanding}. Within any cell $\mathcal{C}_p$, the activation pattern at each hidden layer $l \in \{1, \dots, L-1\}$ is fixed, defined by a diagonal matrix:
\begin{equation}
\Sigma_p^{(l)} = \text{diag}\left(\sigma_{p,1}^{(l)}, \dots, \sigma_{p,h_l}^{(l)}\right), \quad \text{where} \quad \sigma_{p,m}^{(l)} = \begin{cases} 1 & \text{if } z_{p,m}^{(l)} \ge 0 \\ \alpha_{\text{leak}} & \text{if } z_{p,m}^{(l)} < 0 \end{cases}.
\end{equation}
Consequently, within cell $\mathcal{C}_p$, the mapping $f_\theta$ is affine:
\begin{equation}
f_\theta(d) = J_p^\top d + c_p, \quad \text{where} \quad J_p = \left( W^{(L)} \prod_{l=L-1}^1 \Sigma_p^{(l)} W^{(l)} \right)^\top \in \mathbb{R}^k,
\end{equation}
and $c_p \in \mathbb{R}$ is a bias constant.

Because the empirical training data is strictly bounded within $\mathcal{K}_{\text{train}} \subset [0, R_0]^k$, consider an unbounded polyhedral cell $\mathcal{C}_{\text{out}}$ extending to infinity along an arbitrary positive direction $v \in \mathbb{R}_+^k$ ($\mathcal{C}_{\text{out}} \cap \mathcal{K}_{\text{train}} = \emptyset$). The empirical loss objective $\mathcal{L}(\theta; \mathcal{D})$ imposes zero empirical risk constraints on $\mathcal{C}_{\text{out}}$:
\begin{equation}
\frac{\partial \mathcal{L}(\theta)}{\partial J_{\text{out}}} = \mathbf{0} \quad \forall d \in \mathcal{C}_{\text{out}}.
\end{equation}
Under standard Gaussian or He initialization $\theta \sim \mathcal{N}(0, \sigma_{\text{init}}^2 I)$ and stochastic gradient descent (SGD), the marginal distribution of the asymptotic Jacobian elements $(J_{\text{out}})_j$ is symmetric with zero mean:
\begin{equation}
\mathbb{E}[(J_{\text{out}})_j] = 0, \quad \text{Var}((J_{\text{out}})_j) = \sigma_J^2 > 0.
\end{equation}
By symmetry of the unconstrained gradient flow on unpopulated polyhedral cells, the probability that the directional derivative is strictly negative satisfies:
\begin{equation}
\mathbb{P}\left( \inner{J_{\text{out}}}{v} \le -\gamma < 0 \right) = \frac{1}{2} - \Phi\left(\frac{-\gamma}{\sigma_J}\right) > 0.
\end{equation}
Along any ray $d(t) = t \cdot v \in \mathcal{C}_{\text{out}}$ ($t \to \infty$):
\begin{equation}
f_\theta(d(t)) = t \cdot \inner{J_{\text{out}}}{v} + c_{\text{out}} \le -t \cdot \gamma + c_{\text{out}}.
\end{equation}
Taking the asymptotic limit as $t \to \infty$ yields $\lim_{t \to \infty} f_\theta(d(t)) = -\infty$. Applying the sigmoid function $\sigma(z) = (1 + e^{-z})^{-1}$:
\begin{equation}
\lim_{t \to \infty} \sigma(f_\theta(d(t))) = \lim_{t \to \infty} \frac{1}{1 + e^{t \cdot \gamma - c_{\text{out}}}} = 0.0000,
\end{equation}
completing the proof of Part~1.

\textbf{Part 2: Proof of Monotonicity Recovery under MSSP}.
Under MSSP, the set of perturbation scales $\mathcal{S}_{\text{scales}} = \{\epsilon_1, \dots, \epsilon_S\}$ forms a geometric sequence $\epsilon_s = \epsilon_{\min} \cdot \rho^{s-1}$ spanning from $\epsilon_{\min} = 0.2$ to $\epsilon_S = 6.0$. 

Let $\mathcal{A}_s \subset \mathbb{R}^k$ denote the concentric annular distance shell induced by scale $\epsilon_s$:
\begin{equation}
\mathcal{A}_s \triangleq \left\{ d \in \mathbb{R}^k \;\middle|\; \norm{d - D(x_{\text{nom}})}_2 \in [\epsilon_s - \delta_s, \epsilon_s + \delta_s] \right\}.
\end{equation}
By construction, the union of shells covers the unbounded metric domain up to $R_{\max} = \epsilon_S \sqrt{D}$:
\begin{equation}
\bigcup_{s=1}^S \mathcal{A}_s \supseteq [0, R_{\max}]^k.
\end{equation}
For any two adjacent shells $\mathcal{A}_s$ and $\mathcal{A}_{s+1}$, let $d^{(s)} \in \mathcal{A}_s$ and $d^{(s+1)} \in \mathcal{A}_{s+1}$ with coordinate-wise ordering $d^{(s+1)} \ge d^{(s)} + \Delta_d \mathbf{1}$ ($\Delta_d > 0$). Both points are assigned negative training labels $y=1$ with target logit $f^* \ge f_{\text{target}} \gg 0$. 

To minimize binary cross-entropy on all shells $\mathcal{L}(\theta) \le \varepsilon_{\text{opt}}$, the network must output $f_\theta(d) \ge \log(1 / \varepsilon_{\text{opt}} - 1) \triangleq M_{\varepsilon} > 0$ for all $d \in \bigcup_{s=1}^S \mathcal{A}_s$, while outputting $f_\theta(D(x_{\text{nom}})) \le -M_{\varepsilon} < 0$ on nominal references. By the Mean Value Theorem on each active linear cell $\mathcal{C}_p \subset [0, R_{\max}]^k$:
\begin{equation}
f_\theta(d^{(s+1)}) - f_\theta(d^{(s)}) = \inner{\nabla_d f_\theta(\xi)}{d^{(s+1)} - d^{(s)}} \ge 0.
\end{equation}
Since dense synthetic samples anchor the gradient across every step $\Delta_d \le \rho \cdot \epsilon_{\min}$, the Jacobian elements are strictly bounded away from zero:
\begin{equation}
\frac{\partial f_\theta(d)}{\partial d_j} \ge \frac{2 M_{\varepsilon}}{R_{\max}} \triangleq \kappa > 0 \quad \forall j \in \{1, \dots, k\}, \quad \forall d \in [0, R_{\max}]^k.
\end{equation}
Hence, ranking monotonicity is strictly preserved, and out-of-distribution distance inversion is eliminated.
\end{IEEEproof}

\subsection{Theorem 2: Adversarial Negative Gradient Cancellation in Non-IID FL}
\label{subsec:proof_theorem2}

\begin{theorem}[Adversarial Negative Gradient Cancellation]
\label{thm:gradient_cancellation}
Let edge gateways $\mathcal{G}_A$ and $\mathcal{G}_B$ monitor nominal sub-manifolds $\mathcal{M}_A, \mathcal{M}_B \subset \mathbb{R}^D$ satisfying $\mathcal{M}_A \cap \mathcal{M}_B = \emptyset$ with minimum separation $\Delta_{\min} > 0$. Suppose client $\mathcal{G}_A$ generates candidate pseudo-negatives $\tilde{x}_A = x_A + M \odot (\epsilon \cdot z)$ centered on $x_A \in \mathcal{M}_A$ via uncoordinated isotropic perturbation. If the perturbation radius exceeds manifold separation ($\epsilon \sqrt{D \cdot p_{\text{mask}}} > \Delta_{\min}$), then:
\begin{enumerate}
    \item \textbf{(Manifold Intrusion)}: The intrusion set $\Omega_{A \to B} \triangleq (\mathcal{M}_A \oplus \mathcal{B}_D(0, \epsilon)) \cap \mathcal{M}_B$ has strictly positive Lebesgue measure $\lambda(\Omega_{A \to B}) > 0$, inducing an intrusion probability:
    \begin{equation}
    p_{\text{int}} \triangleq \mathbb{P}_{x_A \sim \mathcal{M}_A, z \sim \mathcal{N}}\left( \tilde{x}_A \in \mathcal{M}_B \right) > 0.
    \end{equation}
    \item \textbf{(Opposing Gradient Expectation)}: Under empirical binary cross-entropy loss, the expected inner product between client gradient vectors $\nabla_\theta \mathcal{L}_A$ and $\nabla_\theta \mathcal{L}_B$ is strictly negative:
    \begin{equation}
    \label{eq:thm2_inner_product}
    \mathbb{E}\left[ \inner{\nabla_\theta \mathcal{L}_A}{\nabla_\theta \mathcal{L}_B} \right] \le -p_{\text{int}} \cdot C_{\min} < 0,
    \end{equation}
    where $C_{\min} \triangleq \inf_{x \in \Omega_{A \to B}} \sigma(f_\theta(D(x))) (1 - \sigma(f_\theta(D(x)))) \norm{\nabla_\theta f_\theta(D(x))}_2^2 > 0$.
    \item \textbf{(Gradient Annihilation)}: Under standard Federated Averaging (FedAvg), the aggregated global gradient norm vanishes along the intrusive subspace:
    \begin{equation}
    \norm{\nabla_\theta \mathcal{L}_{\text{global}}}_{\Omega_{A \to B}} = \norm{\frac{1}{2}\left( \nabla_\theta \mathcal{L}_A + \nabla_\theta \mathcal{L}_B \right)}_{\Omega_{A \to B}} \to 0,
    \end{equation}
    paralyzing global model updates and inducing gradient stagnation.
\end{enumerate}
\end{theorem}

\begin{IEEEproof}
We prove each assertion in order:

\textbf{Step 1: Non-Zero Intrusion Measure}.
Let $\mathcal{M}_A$ and $\mathcal{M}_B$ be compact Riemannian sub-manifolds of $\mathbb{R}^D$ separated by Euclidean distance $\Delta_{\min} = \inf_{x_A \in \mathcal{M}_A, x_B \in \mathcal{M}_B} \norm{x_A - x_B}_2 > 0$. The isotropic Gaussian perturbation vector $\xi = M \odot (\epsilon \cdot z)$ has support equal to the linear subspace spanned by mask $M$, with marginal radial density $p_{\norm{\xi}_2}(r)$ governed by a generalized Chi distribution with mean $\mu_r = \epsilon \sqrt{D \cdot p_{\text{mask}}}$. 

Since $\epsilon \sqrt{D \cdot p_{\text{mask}}} > \Delta_{\min}$, the open ball $\mathcal{B}_D(x_A, \epsilon \sqrt{D}) \cap \mathcal{M}_B$ contains a non-empty relative interior. By regularity of the nominal manifold measure $\mu_{\mathcal{M}_B}$, the conditional probability of manifold invasion satisfies:
\begin{equation}
p_{\text{int}} = \int_{\mathcal{M}_A} \left( \int_{\mathcal{M}_B} \mathcal{N}\left(x_B - x_A; \mathbf{0}, \epsilon^2 \text{diag}(M)\right) dx_B \right) d\mathcal{P}_A(x_A) \ge p_{\min} > 0.
\end{equation}

\textbf{Step 2: Derivation of Opposing Gradient Inner Product}.
Let $\tilde{x} \in \Omega_{A \to B}$ be an intrusive sample. For client $\mathcal{G}_A$, sample $\tilde{x}$ is a synthetic negative assigned target label $y_A = 1$. The local gradient of binary cross-entropy loss with respect to parameters $\theta$ is:
\begin{equation}
\label{eq:grad_A_proof}
\nabla_\theta \mathcal{L}_A(\tilde{x}) = \frac{\partial \mathcal{L}_{\text{BCE}}}{\partial f_\theta} \nabla_\theta f_\theta = \left( \sigma(f_\theta(D(\tilde{x}))) - 1 \right) \nabla_\theta f_\theta(D(\tilde{x})).
\end{equation}

Conversely, for client $\mathcal{G}_B$, the identical or infinitesimally close point $x \in \mathcal{M}_B$ is a legitimate nominal packet assigned target label $y_B = 0$. The local gradient for client $\mathcal{G}_B$ is:
\begin{equation}
\label{eq:grad_B_proof}
\nabla_\theta \mathcal{L}_B(x) = \left( \sigma(f_\theta(D(x))) - 0 \right) \nabla_\theta f_\theta(D(x)).
\end{equation}

Since $\tilde{x} \in \mathcal{M}_B$, the distance vectors to the local nominal neighbors of $\mathcal{M}_B$ are identical: $D(\tilde{x}) = D(x)$. Thus, the neural gradient vector is identical: $\nabla_\theta f_\theta(D(\tilde{x})) = \nabla_\theta f_\theta(D(x)) \triangleq g_0(\tilde{x})$. 

Computing the algebraic inner product between equations~\eqref{eq:grad_A_proof} and~\eqref{eq:grad_B_proof}:
\begin{align}
\inner{\nabla_\theta \mathcal{L}_A(\tilde{x})}{\nabla_\theta \mathcal{L}_B(x)} &= \left(\sigma(f_\theta) - 1\right) \left(\sigma(f_\theta)\right) \inner{g_0(\tilde{x})}{g_0(\tilde{x})} \notag \\
&= -\sigma(f_\theta(D(\tilde{x}))) \left(1 - \sigma(f_\theta(D(\tilde{x})))\right) \norm{g_0(\tilde{x})}_2^2.
\end{align}

Because the sigmoid function satisfies $\sigma(z) \in (0, 1)$ for all finite logits, the scalar factor $\sigma(1 - \sigma) > 0$. Taking expectations over the data distributions:
\begin{equation}
\mathbb{E}\left[\inner{\nabla_\theta \mathcal{L}_A}{\nabla_\theta \mathcal{L}_B}\right] \le -p_{\text{int}} \cdot \inf_{x \in \Omega_{A \to B}} \left\{ \sigma(f_\theta) (1 - \sigma(f_\theta)) \norm{g_0(x)}_2^2 \right\} \triangleq -p_{\text{int}} \cdot C_{\min} < 0.
\end{equation}

\textbf{Step 3: Proof of Gradient Annihilation}.
Under FedAvg aggregation with $M=2$ clients:
\begin{equation}
\nabla_\theta \mathcal{L}_{\text{global}}(\tilde{x}) = \frac{1}{2}\left( \nabla_\theta \mathcal{L}_A(\tilde{x}) + \nabla_\theta \mathcal{L}_B(x) \right) = \frac{1}{2}\left( \sigma - 1 + \sigma \right) g_0(\tilde{x}) = \left(\sigma - \frac{1}{2}\right) g_0(\tilde{x}).
\end{equation}
When the distance ranking model is initialized or regularized such that $f_\theta \approx 0$ ($\sigma \approx 0.5$):
\begin{equation}
\norm{\nabla_\theta \mathcal{L}_{\text{global}}(\tilde{x})}_2 = \left| \sigma(f_\theta) - 0.5 \right| \cdot \norm{g_0(\tilde{x})}_2 \to 0,
\end{equation}
completely annihilating the learning signal and freezing the global model parameters along the intersection boundary $\Omega_{A \to B}$.
\end{IEEEproof}

\subsection{Lemma 2.1: Purging Invariance via FSDS and CMNP}
\label{subsec:proof_lemma1}

\begin{lemma}[Purging Invariance]
\label{lem:purging_invariance}
Let peer gateway $\mathcal{G}_j$ broadcast its Federated Subspace Density Sketch $\mathcal{S}_j = (\mu_j, U_j, \Lambda_j, \tau_j, r_j)$. If a candidate pseudo-negative sample $\tilde{x}$ synthesized by client $\mathcal{G}_i$ belongs to the true nominal manifold of peer $\mathcal{G}_j$ ($\tilde{x} \in \mathcal{M}_j$), then Cross-Manifold Negative Purging (CMNP) correctly purges $\tilde{x}$ with probability bounded by:
\begin{equation}
\label{eq:lem1_bound}
\mathbb{P}\left(\text{Purge}(\tilde{x}) == \text{True} \;\middle|\; \tilde{x} \in \mathcal{M}_j \right) \ge 1 - \delta \ge 0.99,
\end{equation}
where $\delta$ is the chi-squared significance level from~\eqref{eq:purging_condition}.
\end{lemma}

\begin{IEEEproof}
By definition of the nominal manifold~\eqref{eq:nominal_manifold}, any legitimate point $x \in \mathcal{M}_j$ satisfies two geometric criteria:
\begin{enumerate}
    \item \textbf{Orthogonal Null-Space Proximity}: $\norm{(I - U_j U_j^\top)(x - \mu_j)}_2 \le \tau_j$.
    \item \textbf{In-Subspace Mahalanobis Boundedness}: $(x - \mu_j)^\top U_j \Lambda_j^{-1} U_j^\top (x - \mu_j) \le \chi^2_{r_j}(1-\delta)$.
\end{enumerate}

Now consider candidate negative $\tilde{x} \in \mathcal{M}_j$. Under the Stage~1 orthogonal test~\eqref{eq:test_residual}:
\begin{equation}
\Delta_{\perp}(\tilde{x}; \mathcal{S}_j) = \norm{(I - U_j U_j^\top)(\tilde{x} - \mu_j)}_2 \le \tau_j,
\end{equation}
which is satisfied with probability $1$ by definition of $\tau_j = \max_{x \in \mathcal{D}_j} \Delta_{\perp}(x)$.

Under the Stage~2 in-subspace test~\eqref{eq:test_mahalanobis}, let $z_j = U_j^\top (\tilde{x} - \mu_j) \in \mathbb{R}^{r_j}$. Because $U_j$ diagonalizes the nominal covariance, the projected coordinates $z_{j,m}$ are uncorrelated with variance $\lambda_{j,m}$. The normalized quadratic form:
\begin{equation}
\mathcal{D}_{\text{Maha}}^2(\tilde{x}; \mathcal{S}_j) = z_j^\top \Lambda_j^{-1} z_j = \sum_{m=1}^{r_j} \left( \frac{z_{j,m}}{\sqrt{\lambda_{j,m}}} \right)^2 \sim \chi^2_{r_j}.
\end{equation}

By properties of the cumulative chi-squared distribution with $r_j$ degrees of freedom:
\begin{equation}
\mathbb{P}\left( \mathcal{D}_{\text{Maha}}^2(\tilde{x}; \mathcal{S}_j) \le \chi^2_{r_j}(1-\delta) \right) = F_{\chi^2_{r_j}}\left(\chi^2_{r_j}(1-\delta)\right) = 1 - \delta.
\end{equation}
Setting significance level $\delta = 0.01$ ensures that invasive candidates are detected and purged with confidence $\ge 1 - 0.01 = 99\%$, establishing the lemma.
\end{IEEEproof}

\subsection{Lemma 2.2: Pareto Directional Convergence via DROGA}
\label{subsec:proof_lemma2}

\begin{lemma}[Pareto Directional Convergence]
\label{lem:pareto_convergence}
Let $g_1, g_2, \dots, g_M \in \mathbb{R}^P$ denote the local gradient vectors computed by $M$ edge gateways, with normalized vectors $\hat{g}_i = g_i / \norm{g_i}_2$. Let $w^* \in \Delta^M$ denote the optimal simplex weights solving the dual quadratic program~\eqref{eq:droga_qp}, and let $g_{\text{aligned}} = \sum_{i=1}^M w_i^* g_i$. Assuming each client loss function $\mathcal{L}_i$ is $L$-Lipschitz smooth ($\norm{\nabla \mathcal{L}_i(\theta_1) - \nabla \mathcal{L}_i(\theta_2)}_2 \le L \norm{\theta_1 - \theta_2}_2$), the aligned update satisfies:
\begin{enumerate}
    \item \textbf{(Non-Negative Directional Alignment)}:
    \begin{equation}
    \label{eq:lem2_non_negative}
    \inner{g_{\text{aligned}}}{g_i} \ge 0 \quad \forall i \in \{1, \dots, M\}.
    \end{equation}
    \item \textbf{(Monotonic Pareto Descent)}: For learning rate $\eta \le \frac{2}{L}$, the federated parameter update $\theta^{(t+1)} \leftarrow \theta^{(t)} - \eta \cdot g_{\text{aligned}}$ guarantees monotonic reduction of the multi-objective loss:
    \begin{equation}
    \label{eq:lem2_descent}
    \mathcal{L}_i(\theta^{(t+1)}) \le \mathcal{L}_i(\theta^{(t)}) - \eta \inner{\nabla \mathcal{L}_i}{g_{\text{aligned}}} + \frac{L \eta^2}{2} \norm{g_{\text{aligned}}}_2^2 \le \mathcal{L}_i(\theta^{(t)}),
    \end{equation}
    preventing client drift and ensuring asymptotic convergence to a Pareto stationary point.
\end{enumerate}
\end{lemma}

\begin{IEEEproof}
We analyze the optimization problem~\eqref{eq:droga_qp}:
\begin{equation}
\min_{w} \frac{1}{2} \norm{\sum_{i=1}^M w_i \hat{g}_i}_2^2 \quad \text{s.t.} \quad \sum_{i=1}^M w_i = 1, \; w_i \ge 0 \; \forall i.
\end{equation}

Let $u^* = \sum_{i=1}^M w_i^* \hat{g}_i$. By the Karush-Kuhn-Tucker (KKT) conditions for convex quadratic programming on a simplex~\cite{boyd2004convex}, there exist Lagrange multipliers $\lambda^* \ge 0$ (for the equality constraint $\sum w_i = 1$) and $\mu_i^* \ge 0$ (for inequality constraints $w_i \ge 0$) such that:
\begin{align}
\nabla_{w_i} \left[ \frac{1}{2} \norm{\sum_{j=1}^M w_j \hat{g}_j}_2^2 - \lambda^* \left(\sum_{j=1}^M w_j - 1\right) - \sum_{j=1}^M \mu_j^* w_j \right] &= 0 \notag \\
\implies \inner{\hat{g}_i}{u^*} - \lambda^* - \mu_i^* &= 0.
\end{align}
By complementary slackness, $\mu_i^* w_i^* = 0$. For any client $i$, since $\mu_i^* \ge 0$:
\begin{equation}
\inner{\hat{g}_i}{u^*} = \lambda^* + \mu_i^* \ge \lambda^*.
\end{equation}
Multiplying by $w_i^*$ and summing across all $i \in \{1, \dots, M\}$:
\begin{equation}
\norm{u^*}_2^2 = \sum_{i=1}^M w_i^* \inner{\hat{g}_i}{u^*} = \lambda^* \sum_{i=1}^M w_i^* + \sum_{i=1}^M \mu_i^* w_i^* = \lambda^* + 0 = \lambda^*.
\end{equation}
Therefore:
\begin{equation}
\inner{\hat{g}_i}{u^*} \ge \norm{u^*}_2^2 \ge 0 \quad \forall i \in \{1, \dots, M\}.
\end{equation}
Substituting $\hat{g}_i = g_i / \norm{g_i}_2$ and $g_{\text{aligned}} = \sum_{j=1}^M w_j^* g_j = \norm{g_j}_{\text{mean}} \cdot u^*$:
\begin{equation}
\inner{g_i}{g_{\text{aligned}}} = \norm{g_i}_2 \cdot \norm{u^*}_2 \cdot \inner{\hat{g}_i}{u^*} \ge 0 \quad \forall i \in \{1, \dots, M\},
\end{equation}
proving equation~\eqref{eq:lem2_non_negative}.

To prove Pareto descent~\eqref{eq:lem2_descent}, by $L$-Lipschitz smoothness of $\mathcal{L}_i$:
\begin{equation}
\mathcal{L}_i(\theta^{(t+1)}) \le \mathcal{L}_i(\theta^{(t)}) + \inner{\nabla \mathcal{L}_i(\theta^{(t)})}{\theta^{(t+1)} - \theta^{(t)}} + \frac{L}{2} \norm{\theta^{(t+1)} - \theta^{(t)}}_2^2.
\end{equation}
Substituting the parameter update $\theta^{(t+1)} - \theta^{(t)} = -\eta \cdot g_{\text{aligned}}$:
\begin{equation}
\mathcal{L}_i(\theta^{(t+1)}) \le \mathcal{L}_i(\theta^{(t)}) - \eta \inner{g_i}{g_{\text{aligned}}} + \frac{L \eta^2}{2} \norm{g_{\text{aligned}}}_2^2.
\end{equation}
For any step-size $\eta \le \frac{2 \inner{g_i}{g_{\text{aligned}}}}{L \norm{g_{\text{aligned}}}_2^2}$, the descent condition $\mathcal{L}_i(\theta^{(t+1)}) \le \mathcal{L}_i(\theta^{(t)})$ holds strictly for every participating client $i$, establishing Pareto stationarity and concluding the proof.
\end{IEEEproof}
